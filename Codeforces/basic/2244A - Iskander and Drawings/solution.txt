import java.util.*;

public class IskanderAndDrawing {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int t = sc.nextInt();

        while (t-- > 0) {
            int n = sc.nextInt();
            String s = sc.next();

            int count = 0;
            int total = 0;

            for (int i = 0; i < n; i++) {
                if (s.charAt(i) == '#') {
                    count++;
                    if(total<count){
                        total=count;
                    }
                } else {
                    count = 0;
                }
            }

            System.out.println((total + 1) / 2);
        }
        sc.close();
    }
}
