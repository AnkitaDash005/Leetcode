import java.util.Scanner;

public class Candles379A {
    public static void main(String[] args) {
        Scanner sc=new Scanner(System.in);
        int a=sc.nextInt();
        int b=sc.nextInt();
        int t=a;
        int left=0;
        while(a>=b){
            left=a%b;
            a/=b;
            t+=a;
            a+=left;
        }
        System.out.println(t);
        sc.close();
    }
}
