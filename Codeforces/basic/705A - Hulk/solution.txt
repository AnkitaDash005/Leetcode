import java.util.*;
public class Hulk_705A {
    public static void main(String[] args) {
        Scanner sc=new Scanner(System.in);
        int n=sc.nextInt();
        String s="";
        for(int i=2;i<=n;i++){
            if(i%2==0){
                s+=" that I love";
            }
            else{
                s+=" that I hate";
            }
        }
        String f="I hate"+s+" it";
        System.out.println(f);
        sc.close();
    }
}
