import java.util.*;

public class NikitaAndBooks {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int t = sc.nextInt();

        while (t-- > 0) {
            int n = sc.nextInt();
            long[] a = new long[n];

            for (int i = 0; i < n; i++) {
                a[i] = sc.nextLong();
            }

            boolean possible = true;

            for (int i = 0; i < n - 1; i++) {
                if (a[i] < i + 1) {
                    possible = false;
                    break;
                }

                long extra = a[i] - (i + 1);
                a[i + 1] += extra;
                a[i] = i + 1;
            }

            if (possible && a[n - 1] < n) {
                possible = false;
            }

            System.out.println(possible ? "YES" : "NO");
        }

        sc.close();
    }
}
