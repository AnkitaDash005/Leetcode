import java.util.*;

public class Divisibility_1328A {
    public static int steps(int a, int b){
        int k=a%b;
        if(k==0){
            return 0;
        }
        return b-k;
    }
    public static void main(String[] args) {
        Scanner sc=new Scanner(System.in);
        int t=sc.nextInt();
        while(t-->0){
            int a=sc.nextInt();
            int b=sc.nextInt();
            System.out.println(steps(a,b));
        }

        sc.close();
    }    
}
