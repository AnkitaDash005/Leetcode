import java.util.*;
public class Lucky {
    public static void main(String[] args) {
        Scanner sc=new Scanner(System.in);
        int b=sc.nextInt();
        while(b-->0){
            int a=sc.nextInt();
            int c=0;
            int c1=0;
            int c2=0;
            int k=6;
            while(k-->0){
                int rem=a%10;
                if(c<3){
                    c1+=rem;
                }
                else{
                    c2+=rem;
                }
                c++;
                a=a/10;
            }
            if(c1==c2){
                System.out.println("YES");
            }
            else{
                System.out.println("NO");
            }
    }
        sc.close();
    }
}
