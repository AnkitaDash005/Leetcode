import java.util.*;

public class Two_gram_977B {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        String s = sc.next();

        HashMap<String, Integer> map = new HashMap<>();

        int max = 0;
        String ans = "";

        for (int i = 0; i < n - 1; i++) {
            String gram = s.substring(i, i + 2);

            int count = map.getOrDefault(gram, 0) + 1;
            map.put(gram, count);

            if (count > max) {
                max = count;
                ans = gram;
            }
        }

        System.out.println(ans);

        sc.close();
    }
}
